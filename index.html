const SHEET_VIAJES = 'Viajes';
const SHEET_REGISTROS = 'Registros';
function doGet(e) {
  if (e.parameter.action === 'getViajes') return getViajes();
  return resp({ok:false});
}
function doPost(e) {
  var data = JSON.parse(e.postData.contents);
  if (data.action === 'saveViaje') return saveViaje(data);
  if (data.action === 'updateEstado') return updateEstado(data);
  if (data.action === 'deleteViaje') return deleteViaje(data);
  if (data.action === 'saveRegistro') return saveRegistroFn(data);
  return resp({ok:false});
}
function resp(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON);
}
function getSheet(name, headers) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sh = ss.getSheetByName(name);
  if (!sh) { sh = ss.insertSheet(name); sh.appendRow(headers); }
  return sh;
}
function saveViaje(d) {
  var sh = getSheet(SHEET_VIAJES,['ID','FECHA','HORA','CONDUCTOR','TRACTO','SEMI','CLIENTE','PRODUCTO','ORIGEN','DESTINO','INTL','OBS','ESTADO']);
  var vals = sh.getDataRange().getValues();
  for (var i=1;i<vals.length;i++) { if (String(vals[i][0])===String(d.id)) return resp({ok:true,dup:true}); }
  sh.appendRow([d.id,d.fecha,d.hora,d.conductor,d.tracto,d.semi||'',d.cliente,d.producto,d.origen,d.destino,d.intl,d.obs||'',d.estado||'EN RUTA']);
  return resp({ok:true});
}
function updateEstado(d) {
  var sh = getSheet(SHEET_VIAJES,['ID','FECHA','HORA','CONDUCTOR','TRACTO','SEMI','CLIENTE','PRODUCTO','ORIGEN','DESTINO','INTL','OBS','ESTADO']);
  var vals = sh.getDataRange().getValues();
  for (var i=1;i<vals.length;i++) { if (String(vals[i][0])===String(d.id)) { sh.getRange(i+1,13).setValue(d.estado); return resp({ok:true}); } }
  return resp({ok:false,error:'not found'});
}
function deleteViaje(d) {
  var sh = getSheet(SHEET_VIAJES,['ID','FECHA','HORA','CONDUCTOR','TRACTO','SEMI','CLIENTE','PRODUCTO','ORIGEN','DESTINO','INTL','OBS','ESTADO']);
  var vals = sh.getDataRange().getValues();
  for (var i=1;i<vals.length;i++) { if (String(vals[i][0])===String(d.id)) { sh.deleteRow(i+1); return resp({ok:true}); } }
  return resp({ok:true});
}
function getViajes() {
  var sh = getSheet(SHEET_VIAJES,['ID','FECHA','HORA','CONDUCTOR','TRACTO','SEMI','CLIENTE','PRODUCTO','ORIGEN','DESTINO','INTL','OBS','ESTADO']);
  var vals = sh.getDataRange().getValues();
  var viajes = [];
  for (var i=1;i<vals.length;i++) {
    if (!vals[i][0]) continue;
    viajes.push({id:vals[i][0],fecha:vals[i][1],hora:vals[i][2],conductor:vals[i][3],tracto:vals[i][4],semi:vals[i][5],cliente:vals[i][6],producto:vals[i][7],origen:vals[i][8],destino:vals[i][9],intl:vals[i][10],obs:vals[i][11],estado:vals[i][12]||'EN RUTA'});
  }
  return resp({ok:true,viajes:viajes});
}
function saveRegistroFn(d) {
  var sh = getSheet(SHEET_REGISTROS,['FECHA REGISTRO','MES','GUIA','FECHA CARGA','FECHA DESCARGA','CONDUCTOR','TRACTO','SEMI','CLIENTE','PRODUCTO','ORIGEN','DESTINO','CANTIDAD','UNIDAD','PESAJE KG','INTL','OBS SALIDA','OBS DESCARGA']);
  var now = new Date();
  sh.appendRow([d.fechaRegistro,d.mes,d.guia,d.fechaCarga,d.fechaDescarga,d.conductor,d.tracto,d.semi,d.cliente,d.producto,d.origen,d.destino,d.cantidad,d.unidad,d.pesaje,d.intl,d.obs,d.obsDescarga]);
  return resp({ok:true});
}
