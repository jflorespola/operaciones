Código Apps Script para copiar
function doPost(e) {
  var sheet = SpreadsheetApp
    .getActiveSpreadsheet()
    .getActiveSheet();

  var data = JSON.parse(e.postData.contents);

  // Header si es la primera fila
  if (sheet.getLastRow() === 0) {
    sheet.appendRow([
      'FECHA REGISTRO','MES','GUIA',
      'FECHA CARGA','FECHA DESCARGA',
      'CONDUCTOR','TRACTO','SEMI',
      'CLIENTE','PRODUCTO',
      'ORIGEN','DESTINO',
      'CANTIDAD','UNIDAD','PESAJE KG',
      'INTL','OBS SALIDA','OBS DESCARGA'
    ]);
  }

  sheet.appendRow([
    data.fechaRegistro, data.mes, data.guia,
    data.fechaCarga, data.fechaDescarga,
    data.conductor, data.tracto, data.semi,
    data.cliente, data.producto,
    data.origen, data.destino,
    data.cantidad, data.unidad, data.pesaje,
    data.intl, data.obs, data.obsDescarga
  ]);

  return ContentService
    .createTextOutput(JSON.stringify({ok:true}))
    .setMimeType(ContentService.MimeType.JSON);
}
